while IFS='=' read -r key raw_value; do
  [[ -z "$key" || "$key" =~ ^# ]] && continue
  # Skip keys with invalid characters for .env
  [[ "$key" =~ [/[:space:]+] ]] && continue
  [[ "$key" =~ ^(DB_|REDIS_|DB=|CONNECTIONSTRINGS__|DATABASE_URL=|LIGHTSAIL_SSH_KEY=|REDIS_HOST=|REDIS_PORT=|REDIS_PASSWORD=) ]] && continue

  # Remove quotes from value
  value="${raw_value%\"}"
  value="${value#\"}"
  value="${value%\'}"
  value="${value#\'}"

  # Convert colons to double underscores for .NET nested keys
  net_key="${key//:/__}"

  echo "${net_key}=${value}" >> /home/***/apps/MenuGreenSystem/.env
done < /tmp/doppler_raw.env
